class Solution {
public:
    int reverseDegree(string s) {
        int result=0;
        for(int i=0;i<s.size();i++){
            int stringindex=i+1;
            int pos=27-(s[i]-96);
            result+=(pos*stringindex);
        }
        return result;
    }
};
