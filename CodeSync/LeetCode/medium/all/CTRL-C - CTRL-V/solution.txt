class Solution {
public:
    int firstStableIndex(vector<int>& nums, int k) {
        int n=nums.size();
        vector<int> revmin(n);
        revmin[n-1]=nums[n-1];
        for(int i=n-2;i>=0;--i){
            revmin[i]=min(nums[i],revmin[i+1]);
        }
    }

};
